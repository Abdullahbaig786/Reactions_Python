from itertools import product #product() creates all possible combinations of items from multiple lists.

# %%
from itertools import product

triplets = []

for catalyst, base, solvent in product(
    recommended_catalysts,
    recommended_bases,
    recommended_solvents
):

    result = df[
        (df["catalyst name"] == catalyst)
        & (df["base name"] == base)
        & (df["solvent name"] == solvent)
    ]

    if not result.empty:
        triplets.append(
            {
                "catalyst": catalyst,
                "base": base,
                "solvent": solvent,
                "area": result["area"].mean()
            }
        )

triplets_df = pd.DataFrame(triplets)

triplets_df.head()

# %%
from itertools import combinations

elo = {
    idx: 1500
    for idx in triplets_df.index
}

K = 32


# %%
for a, b in combinations(triplets_df.index, 2):

    area_a = triplets_df.loc[a, "area"]
    area_b = triplets_df.loc[b, "area"]

    rating_a = elo[a]
    rating_b = elo[b]

    expected_a = 1 / (1 + 10 ** ((rating_b - rating_a) / 400))
    expected_b = 1 / (1 + 10 ** ((rating_a - rating_b) / 400))

    if area_a > area_b:
        score_a = 1
        score_b = 0
    elif area_b > area_a:
        score_a = 0
        score_b = 1
    else:
        score_a = 0.5
        score_b = 0.5

    elo[a] += K * (score_a - expected_a)
    elo[b] += K * (score_b - expected_b)


# %%
triplets_df["elo"] = triplets_df.index.map(elo)

triplets_df.head()

# %%
triplets_df = triplets_df.sort_values(
    "elo",
    ascending=False
)

triplets_df
